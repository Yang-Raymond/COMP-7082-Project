// Install: npx expo install expo-secure-store
// .env:    EXPO_PUBLIC_API_URL=http://<your-computer-LAN-IP>:3000   (not localhost; your phone can't reach that)
import * as SecureStore from 'expo-secure-store';

const API_URL = process.env.EXPO_PUBLIC_API_URL;
const TOKEN_KEY = 'ghostracer_token';
export const isAuthConfigured = Boolean(API_URL);
export type AuthUser = { id: number; username: string; email: string; points: number };
export type Session = { token: string; user: AuthUser };

const listeners = new Set<(s: Session | null) => void>();
const notify = (s: Session | null) => listeners.forEach((cb) => cb(s));

async function api<T>(path: string, body?: unknown, token?: string): Promise<T> {
  if (!API_URL) throw new Error('Login is not configured. Add EXPO_PUBLIC_API_URL to your .env file.');
  try {
    const res = await fetch(`${API_URL}${path}`, {
      method: body ? 'POST' : 'GET',
      headers: {
        'Content-Type': 'application/json',
        ...(token ? { Authorization: `Bearer ${token}` } : {}),
      },
      body: body ? JSON.stringify(body) : undefined,
    });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) throw new Error(data.error ?? `Request failed (${res.status})`);
    return data as T;
  } catch (e) {
    const msg = e instanceof Error ? e.message : String(e);
    throw new Error(`${msg} [${API_URL}${path}]`);
  }
}

export async function signUp(
  username: string,
  email: string,
  password: string
): Promise<{ needsConfirmation: boolean }> {
  const session = await api<Session>('/auth/signup', { username, email, password });
  await SecureStore.setItemAsync(TOKEN_KEY, session.token);
  notify(session);
  return { needsConfirmation: false };
}

export async function signIn(email: string, password: string): Promise<void> {
  const session = await api<Session>('/auth/login', { email, password });
  await SecureStore.setItemAsync(TOKEN_KEY, session.token);
  notify(session);
}

export async function signOut(): Promise<void> {
  await SecureStore.deleteItemAsync(TOKEN_KEY);
  notify(null);
}

export async function getSession(): Promise<Session | null> {
  const token = await SecureStore.getItemAsync(TOKEN_KEY);
  if (!token) return null;
  try {
    const { user } = await api<{ user: AuthUser }>('/auth/me', undefined, token);
    return { token, user };
  } catch (e) {
    const m = (e instanceof Error ? e.message : '').toLowerCase();
    // Only drop the token if the server rejected it, not if we're just offline
    if (m.includes('invalid login credentials')) await SecureStore.deleteItemAsync(TOKEN_KEY);
    return null;
  }
}

export function onAuthChange(callback: (session: Session | null) => void): () => void {
  listeners.add(callback);
  return () => {
    listeners.delete(callback);
  };
}

/** Emails a 6-digit reset code. Always succeeds, so it can't be used to check which emails exist. */
export async function requestPasswordReset(email: string): Promise<void> {
  await api<{ ok: boolean }>('/auth/forgot-password', { email: email.trim() });
}

/** Checks the emailed code and sets the new password. */
export async function resetPassword(email: string, code: string, newPassword: string): Promise<void> {
  await api<{ ok: boolean }>('/auth/reset-password', {
    email: email.trim(),
    code: code.trim(),
    password: newPassword,
  });
}

export function friendlyAuthError(e: unknown): string {
  const msg = e instanceof Error ? e.message : String(e);
  const m = msg.toLowerCase();
  if (m.includes('invalid login credentials')) return 'Wrong email or password.';
  if (m.includes('already registered')) return 'An account with this email or username already exists. Try signing in.';
  if (m.includes('network request failed') || m.includes('failed to fetch'))
    return "Can't reach the server. Check your connection.";
  if (m.includes('password should be')) return 'Password is too weak. Use at least 6 characters.';
  if (m.includes('invalid or expired code')) return 'That code is wrong or has expired. Request a new one.';
  return msg;
}

import React, { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';
import type { AuthUser, Session } from './auth';
import { useAuth } from './useAuth';

interface AuthContextValue {
  session: Session | null;
  user: AuthUser | null;
  isSignedIn: boolean;
 
  canChallenge: boolean;
  loading: boolean;
  configured: boolean;
  requireAuth: (reason: string, action: () => void) => void;
  /** Non-null while the "please sign in" prompt should be showing. */
  prompt: { reason: string } | null;
  dismissPrompt: () => void;
}

const AuthContext = createContext<AuthContextValue | null>(null);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const { session, loading, configured } = useAuth();
  const [prompt, setPrompt] = useState<{ reason: string } | null>(null);

  const isSignedIn = Boolean(session);
  const canChallenge = isSignedIn || !configured;

  const requireAuth = useCallback(
    (reason: string, action: () => void) => {
      if (canChallenge) {
        action();
      } else {
        setPrompt({ reason });
      }
    },
    [canChallenge]
  );

  // Signing in closes any open prompt
  useEffect(() => {
    if (isSignedIn) setPrompt(null);
  }, [isSignedIn]);

  const value = useMemo<AuthContextValue>(
    () => ({
      session,
      user: session?.user ?? null,
      isSignedIn,
      canChallenge,
      loading,
      configured,
      requireAuth,
      prompt,
      dismissPrompt: () => setPrompt(null),
    }),
    [session, isSignedIn, canChallenge, loading, configured, requireAuth, prompt]
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuthContext(): AuthContextValue {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuthContext must be used inside <AuthProvider>');
  return ctx;
}

import { useEffect, useState } from 'react';
import type { Session } from './auth';
import { getSession, isAuthConfigured, onAuthChange } from './auth';

export function useAuth() {
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(isAuthConfigured);

  useEffect(() => {
    if (!isAuthConfigured) return;
    let active = true;
    getSession().then((s) => {
      if (!active) return;
      setSession(s);
      setLoading(false);
    });
    const unsubscribe = onAuthChange(setSession);
    return () => {
      active = false;
      unsubscribe();
    };
  }, []);

  return { session, loading, configured: isAuthConfigured };
}

function Shell() {

  const [tab, setTab] = useState<Tab>('home');
  const [returnTab, setReturnTab] = useState<Tab | null>(null); 
  const [raceRoute, setRaceRoute] = useState<SavedRoute | null>(null);
  
  const rec = useRouteRecorder();
  const [draft, setDraft] = useState<RouteDraft | null>(null);

  useEffect(() => {
    if (session && returnTab) {
      setTab(returnTab);
      setReturnTab(null);
    }
  }, [session, returnTab]);

        <View style={{ flex: 1, alignItems: 'center', justifyContent: 'center' }}>

        </View>
      </PhoneFrame>
    );
  }

  const changeTab = (t: Tab) => {
    setRaceRoute(null); 
    setReturnTab(null);
    setTab(t);
  };

  const goSignIn = () => {
    setReturnTab(tab);
    dismissPrompt();
    setRaceRoute(null);
    setTab('you');
  };

  let content: React.ReactNode;
  if (raceRoute) {
    
    content = <RaceScreen route={raceRoute} rec={rec} onExit={() => setRaceRoute(null)} />;

    content = <PlaceholderScreen title="Social" />;

    content = <YouScreen />;

      <View style={{ flex: 1 }}>{content}</View>
      <TabBar tab={tab} onChange={changeTab} />
      {prompt && <SignInPrompt reason={prompt.reason} onSignIn={goSignIn} onCancel={dismissPrompt} />}
    </PhoneFrame>
  );
}

    expanded: expandedH,
  };

  
  const dims = useRef({ expandedH, visible });
  dims.current = { expandedH, visible };
  const y = (s: Snap) => dims.current.expandedH - dims.current.visible[s];
  const clampY = (v: number) => Math.min(Math.max(v, y('expanded')), y('collapsed'));

  const snapTo = (s: Snap) => {
    snapRef.current = s;
    setSnap(s);
    Animated.spring(ty, { toValue: y(s), friction: 9, tension: 70, useNativeDriver: native }).start();
  };

  // Place the sheet once we know the container size (and again if it resizes)
  useEffect(() => {
    if (H > 0) ty.setValue(y(snapRef.current));
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [H]);

  const pan = useRef(
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onMoveShouldSetPanResponder: () => true,
      onPanResponderGrant: () => {
        ty.stopAnimation((v) => {
          startY.current = v;
        });
      },
      onPanResponderMove: (_, g) => ty.setValue(clampY(startY.current + g.dy)),
      onPanResponderRelease: (_, g) => {
        // A tap (barely moved) steps to the next position
        if (Math.abs(g.dx) < 5 && Math.abs(g.dy) < 5) {
          snapTo(ORDER[(ORDER.indexOf(snapRef.current) + 1) % ORDER.length]);
          return;
        }
        // A drag settles on the nearest position, nudged by flick speed
        const projected = clampY(startY.current + g.dy) + g.vy * 150;
        const best = ORDER.reduce((a, b) => (Math.abs(y(a) - projected) < Math.abs(y(b) - projected) ? a : b));
        snapTo(best);
      },
    })
  ).current;

  return (
    <View
      style={StyleSheet.absoluteFill}
      pointerEvents="box-none"   
      onLayout={(e) => setH(e.nativeEvent.layout.height)}
    >
      {H > 0 && (
        <Animated.View style={[s.sheet, shadow, { height: expandedH, transform: [{ translateY: ty }] }]}>
          <View {...pan.panHandlers} style={s.header}>
            <View style={s.handle} />
            <View style={s.titleRow}>
              <Text style={s.title}>{title}</Text>
            </View>
          </View>
          <ScrollView
            style={{ height: expandedH - HEADER }}
            contentContainerStyle={{ paddingHorizontal: 16, gap: 12, paddingBottom: expandedH - visible[snap] + 16 }}
            showsVerticalScrollIndicator={false}
          >
            {children}
          </ScrollView>
        </Animated.View>
      )}

  );

export interface RouteDraft {
  points: RoutePoint[];
  durationMs: number;
  distanceMeters: number;
  elevationGainMeters: number;
  activity: ActivityType;
}

export function useRouteRecorder() {
  const [status, setStatus] = useState<RecorderStatus>('idle');
  const [activity, setActivity] = useState<ActivityType>('run');
  const [points, setPoints] = useState<RoutePoint[]>([]);
  const [elapsedMs, setElapsedMs] = useState(0);

  const trackerRef = useRef(new GpsTracker());
  const statusRef = useRef<RecorderStatus>('idle');
  const accumulatedRef = useRef(0); 
  const segmentStartRef = useRef(0);

  const changeStatus = (s: RecorderStatus) => {
    statusRef.current = s;
    setStatus(s);
  };

  
  const activeMs = () =>
    accumulatedRef.current +
    (statusRef.current === 'recording' ? Date.now() - segmentStartRef.current : 0);

  useEffect(() => {
    if (status !== 'recording') return;
    const id = setInterval(() => setElapsedMs(activeMs()), 500);
    return () => clearInterval(id);
  }, [status]);

  const start = useCallback(async () => {
    setPoints([]);
    setElapsedMs(0);
    accumulatedRef.current = 0;
    segmentStartRef.current = Date.now();
    changeStatus('recording');

    await trackerRef.current.start(
      (fix) => {
        if (statusRef.current !== 'recording') return;
        setPoints((prev) => [
          ...prev,
          { lat: fix.lat, lng: fix.lng, t: activeMs(), alt: fix.altitude ?? undefined },
        ]);
      },
      (message) => {
        console.warn('Recorder GPS error:', message);
        changeStatus('idle'); 
      }
    );
  }, []);

  const pause = useCallback(() => {
    accumulatedRef.current = activeMs();
    changeStatus('paused');
    setElapsedMs(accumulatedRef.current);
  }, []);

  const resume = useCallback(() => {
    segmentStartRef.current = Date.now();
    changeStatus('recording');
  }, []);

  const distanceMeters = useMemo(
    () => totalDistance(points.map((p) => ({ lat: p.lat, lng: p.lng }))),
    [points]
  );

  
  const elevationGainMeters = useMemo(() => elevationGain(points), [points]);

  const finish = useCallback((): RouteDraft => {
    const durationMs = activeMs();
    trackerRef.current.stop();
    changeStatus('idle');
    return { points, durationMs, distanceMeters, elevationGainMeters, activity };
  }, [points, distanceMeters, elevationGainMeters, activity]);

  return {
    status,
    activity,
    setActivity,
    points,
    elapsedMs,
    distanceMeters,
    elevationGainMeters,
    start,
    pause,
    resume,
    finish,
  };
}

interface LatLng {
  lat: number;
  lng: number;
}

const EARTH_RADIUS_M = 6371000;

/** Great-circle distance between two points, in meters. */
export function haversineDistance(a: LatLng, b: LatLng): number {
  const toRad = (deg: number) => (deg * Math.PI) / 180;
  const dLat = toRad(b.lat - a.lat);
  const dLng = toRad(b.lng - a.lng);

  const sinLat = Math.sin(dLat / 2);
  const sinLng = Math.sin(dLng / 2);

  const h =
    sinLat * sinLat +
    Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * sinLng * sinLng;

  return 2 * EARTH_RADIUS_M * Math.asin(Math.sqrt(h));
}

/** Running total distance across an ordered list of points. */
export function totalDistance(points: LatLng[]): number {
  let sum = 0;
  for (let i = 1; i < points.length; i++) {
    sum += haversineDistance(points[i - 1], points[i]);
  }
  return sum;
}

export function formatDistance(meters: number): string {
  return meters < 1000 ? `${Math.round(meters)} m` : `${(meters / 1000).toFixed(2)} km`;
}

export function elevationGain(points: { alt?: number }[]): number {
  let gain = 0;
  let prev: number | undefined;
  for (const p of points) {
    if (p.alt == null) continue;
    if (prev == null) {
      prev = p.alt;
      continue;
    }
    const d = p.alt - prev;
    if (Math.abs(d) >= 2) {
      if (d > 0) gain += d;
      prev = p.alt;
    }
  }
  return gain;
}

import { SavedRoute, GhostDelta } from '../types/route';
import { haversineDistance } from './distance';

export function computeGhostDelta(
  route: SavedRoute,
  currentLat: number,
  currentLng: number,
  elapsedMs: number
): GhostDelta | null {
  if (route.points.length === 0) return null;

  let closestIndex = 0;
  let closestDist = Infinity;

  for (let i = 0; i < route.points.length; i++) {
    const p = route.points[i];
    const d = haversineDistance({ lat: currentLat, lng: currentLng }, { lat: p.lat, lng: p.lng });
    if (d < closestDist) {
      closestDist = d;
      closestIndex = i;
    }
  }

  const ghostPoint = route.points[closestIndex];

  return {
    deltaMs: elapsedMs - ghostPoint.t,
    progress: closestIndex / (route.points.length - 1 || 1),
    ghostPosition: { lat: ghostPoint.lat, lng: ghostPoint.lng },
  };
}

export function formatDelta(deltaMs: number): string {
  const seconds = Math.round(Math.abs(deltaMs) / 1000);
  const sign = deltaMs < 0 ? 'ahead' : 'behind';
  if (seconds < 60) return `${seconds}s ${sign}`;
  const mins = Math.floor(seconds / 60);
  const secs = seconds % 60;
  return `${mins}m ${secs}s ${sign}`;
}

import * as Location from "expo-location";
import { GpsFix } from "../types/gps";

export type FixListener = (fix: GpsFix) => void;
export type ErrorListener = (message: string) => void;

function toGpsFix(loc: Location.LocationObject): GpsFix {
  const c = loc.coords;
  return {
    lat: c.latitude,
    lng: c.longitude,
    accuracy: c.accuracy,
    speed: c.speed,
    heading: c.heading,
    altitude: c.altitude,
    timestamp: loc.timestamp,
  };
}

export class GpsTracker {
  private subscription: Location.LocationSubscription | null = null;

  
  async requestPermission(): Promise<boolean> {
    const { status } = await Location.requestForegroundPermissionsAsync();
    return status === "granted";
  }

  async start(onFix: FixListener, onError: ErrorListener): Promise<void> {
    if (this.subscription) return; // already tracking

    const granted = await this.requestPermission();
    if (!granted) {
      onError("Location permission denied");
      return;
    }

    try {
      this.subscription = await Location.watchPositionAsync(
        {
          accuracy: Location.Accuracy.BestForNavigation,
          timeInterval: 1000, 
          distanceInterval: 1, 
        },
        (loc) => onFix(toGpsFix(loc))
      );
    } catch (err) {
      onError(err instanceof Error ? err.message : "Unknown GPS error");
    }
  }

  
  async getCurrent(): Promise<GpsFix | null> {
    const granted = await this.requestPermission();
    if (!granted) return null;
    try {
      const loc = await Location.getCurrentPositionAsync({
        accuracy: Location.Accuracy.Balanced,
      });
      return toGpsFix(loc);
    } catch {
      return null;
    }
  }

  stop(): void {
    this.subscription?.remove();
    this.subscription = null;
  }

  get isTracking(): boolean {
    return this.subscription !== null;
  }
}
